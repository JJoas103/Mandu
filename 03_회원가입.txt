#2 userRouter 생성
    //회원가입 페이지
    router.get('/join', (req, res) => {
        res.render('user/join');
    });

    //회원가입 처리
    router.post('/join', async (req, res) => {

        try {
            const { email, password, name, address } = req.body;
            //회원 생성
            const newUser = new User({
                email,
                password,
                name,
                address
            });
            await newUser.save();
            console.log('회원가입 완료');
        } catch (error) {
            console.error('가입 실패', error.message);
        }
    });
