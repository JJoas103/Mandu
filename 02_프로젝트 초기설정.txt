#1 app.js 초기 설정 코드
    // 패키지 불러오기
    const express = require('express');
    const path = require('path');

    // app 생성
    const app = express();

    // 뷰 엔진 설정
    app.set('view engine', 'ejs');
    app.set('views', path.join(__dirname, "views"));

    // 미들웨어
    app.use(express.static(path.join(__dirname, 'public')));//정적 파일

    //라우터 등록
    app.get('/', (req, res)=>{
        res.render('index', {title : '메인페이지'});
    });

    //서버 시작
    app.listen(3000, ()=>{
        console.log(`3000번 포트에서 서버가 실행 중입니다`)
    });


#2 헤더/푸터 분리
== views > index.ejs 생성
- index.html 코드 복사 후 index.ejs에 불여넣기
- views > partials > header.ejs / footer.ejs 만들기

== 헤더 만들기
- index.ejs에서 헤더부분 잘라내기 후 header.ejs에 붙어넣기    
	<!-- 헤더 -->
    <nav class="navbar navbar-expand-lg navbar-dark">
		<div class="container">
			<a class="navbar-brand" href="index.html">솔데스크</a>
			<button class="navbar-toggler" type="button" data-bs-toggle="collapse"
				data-bs-target="#nav">
				<span class="navbar-toggler-icon"></span>
			</button>
			<div class="collapse navbar-collapse" id="nav">
				<ul class="navbar-nav ms-auto">
					<li class="nav-item"><a class="nav-link" href="member_info.html">마이페이지</a></li>
					<li class="nav-item"><a class="nav-link" href="member_login.html">로그아웃</a></li>
					<li class="nav-item"><a class="nav-link" href="member_login.html">로그인</a></li>
					<li class="nav-item"><a class="nav-link" href="member_join.html">회원가입</a></li>
				</ul>
			</div>
		</div>
	</nav>

- index.ejs: 헤더가 있던 자리에 header.ejs를 include
    <!-- 헤더 -->
    <%- include('partials/header') %>
    ...

- index.ejs에서 푸터부분 잘라내기 후 footer.ejs에 붙어넣기 
    <!-- 푸터 -->
    <footer class="py-5 mt-5">
        <div class="container text-center footer">
            <div class="small">© 2025 솔데스크 커뮤니티 게시판</div>
        </div>
    </footer>

- index.ejs: 헤더가 있던 자리에 footer.ejs를 include
    <!-- 푸터 -->
    <%- include('partials/footer') %>
    ...
* 다른 HTML 파일들도 ejs 전환 후 위와 동일하게 header/footer 나누기
